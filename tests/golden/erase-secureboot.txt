==> Adapter: arch
==> Step 1: Validate plan and re-probe the disk
lsblk --json -o NAME,SIZE,MODEL,MOUNTPOINT /dev/disk/by-id/nvme-EXAMPLE_SERIAL
check the repositories in /etc/pacman.conf are reachable
==> Step 2: Partition the disk
sfdisk --wipe always /dev/disk/by-id/nvme-EXAMPLE_SERIAL
    <<< label: gpt
    <<< size=1GiB, type=uefi, name="ESP"
    <<< type=linux, name="root"
partprobe /dev/disk/by-id/nvme-EXAMPLE_SERIAL
udevadm settle
==> Step 3: Format and create btrfs subvolumes
mkfs.fat -F32 -n ESP /dev/disk/by-id/nvme-EXAMPLE_SERIAL-part1
mkfs.btrfs -f -L root /dev/disk/by-id/nvme-EXAMPLE_SERIAL-part2
mkdir -p /mnt/target
mount /dev/disk/by-id/nvme-EXAMPLE_SERIAL-part2 /mnt/target
btrfs subvolume create /mnt/target/@
btrfs subvolume create /mnt/target/@home
umount /mnt/target
==> Step 4: Mount the target
mount -o subvol=@,compress=zstd /dev/disk/by-id/nvme-EXAMPLE_SERIAL-part2 /mnt/target
mkdir -p /mnt/target/home
mount -o subvol=@home,compress=zstd /dev/disk/by-id/nvme-EXAMPLE_SERIAL-part2 /mnt/target/home
mkdir -p /mnt/target/boot
mount /dev/disk/by-id/nvme-EXAMPLE_SERIAL-part1 /mnt/target/boot
==> Step 5: Install the base system
pacstrap -K -C /etc/pacman.conf /mnt/target luminos-base luminos-desktop
gpgconf --homedir /mnt/target/etc/pacman.d/gnupg --kill all
==> Step 7: Write fstab, locale, keymap, timezone, hostname, zram
genfstab -U /mnt/target  # stdout appended to /mnt/target/etc/fstab
uncomment "en_US.UTF-8 UTF-8" in /mnt/target/etc/locale.gen
arch-chroot /mnt/target locale-gen
write /mnt/target/etc/locale.conf
    <<< LANG=en_US.UTF-8
write /mnt/target/etc/vconsole.conf
    <<< KEYMAP=us
symlink /mnt/target/etc/localtime -> /usr/share/zoneinfo/Europe/Berlin
write /mnt/target/etc/hostname
    <<< ada-laptop
write /mnt/target/etc/systemd/zram-generator.conf
    <<< [zram0]
    <<< zram-size = ram / 2
    <<< compression-algorithm = zstd
write /mnt/target/etc/skel/.config/hypr/keyboard.conf
    <<< input {
    <<<     kb_layout = us
    <<<     kb_variant = 
    <<< }
==> Step 8: Create the user, lock root, enable sudo
arch-chroot /mnt/target useradd -m -G wheel -c 'Ada Lovelace' -s /bin/bash ada
arch-chroot /mnt/target chpasswd
    <<< [REDACTED]
arch-chroot /mnt/target passwd -l root
write /mnt/target/etc/sudoers.d/10-wheel
    <<< %wheel ALL=(ALL:ALL) ALL
==> Step 9: Build UKIs
write /mnt/target/etc/kernel/cmdline
    <<< root=/dev/disk/by-id/nvme-EXAMPLE_SERIAL-part2 rootflags=subvol=@ rw console=tty0 console=ttyS0,115200n8
write /mnt/target/etc/mkinitcpio.d/linux.preset
    <<< # mkinitcpio preset for the linux package — built by dawn-backend
    <<< 
    <<< ALL_kver="/boot/vmlinuz-linux"
    <<< 
    <<< PRESETS=('default' 'fallback')
    <<< 
    <<< default_uki="/boot/EFI/Linux/arch-linux.efi"
    <<< default_options="--cmdline /etc/kernel/cmdline"
    <<< 
    <<< fallback_uki="/boot/EFI/Linux/arch-linux-fallback.efi"
    <<< fallback_options="-S autodetect --cmdline /etc/kernel/cmdline"
mkdir -p /mnt/target/boot/EFI/Linux
arch-chroot /mnt/target mkinitcpio -P
==> Step 10: Secure Boot: create keys, sign, enroll
arch-chroot /mnt/target sbctl create-keys
arch-chroot /mnt/target sbctl sign -s /boot/EFI/systemd/systemd-bootx64.efi
arch-chroot /mnt/target sbctl sign -s /boot/EFI/Linux/arch-linux.efi
arch-chroot /mnt/target sbctl sign -s /boot/EFI/Linux/arch-linux-fallback.efi
arch-chroot /mnt/target sbctl enroll-keys --microsoft
==> Step 11: Install the bootloader
arch-chroot /mnt/target bootctl install
==> Step 12: Enable services, copy the log, unmount
arch-chroot /mnt/target systemctl enable NetworkManager
arch-chroot /mnt/target systemctl enable greetd
set [initial_session] in /mnt/target/etc/greetd/config.toml
    <<< command = "Hyprland"
    <<< user = "ada"
cp /var/log/dawn.log /mnt/target/var/log/dawn.log
umount -R /mnt/target
sync
